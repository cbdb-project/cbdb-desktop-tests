# Test-suite dependencies.  Deliberately tiny: the application under test
# is a self-contained Go binary plus a SQLite file, so the suite needs
# only an HTTP client and pytest.  Staging, .env parsing and the SQLite
# oracle all use the standard library.

pytest>=7.4,<9
pytest-json-report>=1.5,<2     # machine-readable run record for reports/
requests>=2.31,<3              # HTTP client for the running cbdb.exe

# -- reports --------------------------------------------------------------
# The issue reports are written in English and Traditional Chinese, as
# Markdown, Word and PDF.  python-docx writes the .docx; the .pdf is
# produced by Word itself through COM, because these documents are
# bilingual and Word already has the CJK fonts and line-breaking rules.
# Without pywin32 (or without Word) the .md and .docx are still written
# and the generator reports which formats it could not produce.
python-docx>=1.1,<2
pywin32>=305 ; sys_platform == "win32"
py7zr>=1.0

# Drives the shipped pages in a real Chromium, for the defects that live
# in the pages' own JavaScript and are invisible over HTTP: a control
# that stays greyed out with its precondition met, a page that throws
# during load.  Needs its own browser download:
#
#     python -m playwright install chromium
#
# Without it, tests/test_ui_pages.py skips with that instruction rather
# than failing -- a suite that goes red on a missing optional browser
# teaches people to ignore it.
playwright>=1.40
